function closestVector(tbl,vec)
    local ASD = 99999999
    local closest = vec
    
    for i,v in pairs(tbl) do
        if v:getDistance(vec)<ASD then
            ASD = v:getDistance(vec)
            closest = v
        end
    end
    return closest
end

function buildGraph(points, maxDist)
    local graph = {}

    for i = 1, #points do
        graph[i] = {}

        for j = 1, #points do
            if i ~= j then
                local dist = points[i]:getDistance(points[j])

                if trace.line(points[i], points[j],table.add({chip(),find.byModel("models/props_phx/games/chess/white_knight.mdl")[1]},find.byModel("models/props_phx/games/chess/white_pawn.mdl"))).Hit == false then
                    graph[i][j] = dist
                end
            end
        end
    end

    return graph
end

function getPath(prev, target)
    local path = {}

    while target do
        table.insert(path, 1, target)
        target = prev[target]
    end

    return path
end

function Dijkstra(graph, source)
    local dist = {}
    local prev = {}
    local Q = {}

    for v, _ in pairs(graph) do
        dist[v] = 99999999999
        prev[v] = nil
        table.insert(Q, v)
    end

    dist[source] = 0

    while table.count(Q) > 0 do
        local uIndex = 1
        local u = Q[1]

        for i = 2, #Q do
            if dist[Q[i]] < dist[u] then
                u = Q[i]
                uIndex = i
            end
        end

        table.remove(Q, uIndex)

        for v, cost in pairs(graph[u]) do
            local alt = dist[u] + cost

            if alt < dist[v] then
                dist[v] = alt
                prev[v] = u
            end
        end
    end

    return dist, prev
end
